cmake_minimum_required(VERSION 3.25)
project(thumbnail-consumer LANGUAGES CXX)
if(ISOLATED)
  find_package(HolonightThumbnails 0.1 CONFIG REQUIRED COMPONENTS Isolated)
else()
  find_package(HolonightThumbnails 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE HolonightThumbnails::Thumbnails)

if(ISOLATED)
  target_link_libraries(consumer PRIVATE HolonightThumbnails::Isolated)
  target_compile_definitions(consumer PRIVATE ISOLATED)
endif()
